    p(x, ...)

    p(x, margin = NULL, na.rm = FALSE)

    # S4 method for discrete
    p(x, ..., distribution = NULL, margin = NULL, na.rm = TRUE)

    # S4 method for numeric
    p(x, density = NULL, na.rm = FALSE, ..., bw = "SJ", adjust = 1.5)

    # S4 method for missing
    p(x, ..., margin = NULL, na.rm = TRUE)
